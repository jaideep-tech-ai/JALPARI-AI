    79→// ============ Password field with strength + visibility ============
    80→function PasswordField({
    81→  value, onChange, placeholder, show, setShow, autoComplete,
    82→}: {
    83→  value: string; onChange: (v: string) => void; placeholder: string;
    84→  show: boolean; setShow: (v: boolean) => void; autoComplete?: string;
    85→}) {
    86→  const t = useTranslations('auth');
    87→  const strength = value.length >= 12 && /[A-Z]/.test(value) && /[0-9]/.test(value) ? 4
    88→    : value.length >= 8 && /[A-Za-z]/.test(value) && /[0-9]/.test(value) ? 3
    89→    : value.length >= 8 ? 2 : value.length >= 4 ? 1 : 0;
    90→  const labels = [t('pwdWeak'), t('pwdWeak'), t('pwdFair'), t('pwdGood'), t('pwdStrong')];
    91→  const colors = ['bg-red-500', 'bg-red-500', 'bg-amber-500', 'bg-sky-500', 'bg-emerald-500'];
    92→  return (
    93→    <div>
    94→      <div className="relative">
    95→        <Input
    96→          type={show ? 'text' : 'password'}
    97→          value={value}
    98→          onChange={(e) => onChange(e.target.value)}
    99→          placeholder={placeholder}
   100→          autoComplete={autoComplete}
   101→          className={inputCls + ' pr-10'}
   102→        />
   103→        <button
   104→          type="button"
   105→          onClick={() => setShow(!show)}
   106→          className="absolute right-2.5 top-2.5 text-slate-400 hover:text-white"
   107→          aria-label={show ? t('hidePassword') : t('showPassword')}
   108→        >
   109→          {show ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   110→        </button>
   111→      </div>
   112→      {value && (
   113→        <div className="mt-1.5 flex items-center gap-2">
   114→          <div className="flex gap-1">
   115→            {[1, 2, 3, 4].map((i) => (
   116→              <span key={i} className={`h-1 w-8 rounded-full ${i <= strength ? colors[strength] : 'bg-white/10'}`} />
   117→            ))}
   118→          </div>
   119→          <span className="text-[11px] text-slate-400">{labels[strength]}</span>
   120→        </div>
   121→      )}
   122→    </div>
   123→  );
   124→}

   242→// ============ Sign up ============
   243→export function SignUpScreen() {
   244→  const t = useTranslations('auth');
   245→  const tv = useTranslations('auth.validation');
   246→  const { setView, setPending } = useAppStore();
   247→  const [name, setName] = useState('');
   248→  const [email, setEmail] = useState('');
   249→  const [password, setPassword] = useState('');
   250→  const [confirm, setConfirm] = useState('');
   251→  const [language, setLanguage] = useState<'en' | 'hi'>('en');
   252→  const [organization, setOrganization] = useState('');
   253→  const [terms, setTerms] = useState(false);
   254→  const [show, setShow] = useState(false);
   255→  const [loading, setLoading] = useState(false);
   256→  const [err, setErr] = useState('');
   257→
   258→  async function submit(e: React.FormEvent) {
   259→    e.preventDefault();
   260→    setErr('');
   261→    if (name.trim().length < 2) return setErr(tv('nameMin'));
   262→    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return setErr(tv('emailInvalid'));
   263→    if (password.length < 8) return setErr(tv('pwdMin'));
   264→    if (password !== confirm) return setErr(tv('pwdMatch'));
   265→    if (!terms) return setErr(tv('termsRequired'));
   266→    setLoading(true);
   267→    try {
   268→      const res = await api<{ demoVerifyToken: string }>('/api/auth', {
   269→        body: { action: 'signup', name, email, password, language, organization },
   270→      });
   271→      setPending({ email, token: res.demoVerifyToken });
   272→      setView('verify');
   273→    } catch (e2) {
   274→      setErr((e2 as Error).message);
   275→    } finally {
   276→      setLoading(false);
   277→    }
   278→  }
   279→
   280→  return (
   281→    <AuthShell title={t('signUpTitle')} sub={t('signUpSub')}>
   282→      <form onSubmit={submit} className="space-y-4">
   283→        <ErrAlert msg={err} />
   284→        <div className="space-y-1.5">
   285→          <Label htmlFor="su-name" className="text-slate-300">{t('fullName')}</Label>
   286→          <Input id="su-name" value={name} onChange={(e) => setName(e.target.value)} className={inputCls} autoComplete="name" />
   287→        </div>
   288→        <div className="space-y-1.5">
   289→          <Label htmlFor="su-email" className="text-slate-300">{t('email')}</Label>
   290→          <Input id="su-email" type="email" value={email} onChange={(e) => setEmail(e.target.value)} className={inputCls} autoComplete="email" />
   291→        </div>
   292→        <div className="grid gap-4 sm:grid-cols-2">
   293→          <div className="space-y-1.5">
   294→            <Label className="text-slate-300">{t('preferredLanguage')}</Label>
   295→            <Select value={language} onValueChange={(v) => setLanguage(v as 'en' | 'hi')}>
   296→              <SelectTrigger className={inputCls}><SelectValue /></SelectTrigger>
   297→              <SelectContent>
   298→                <SelectItem value="en">English</SelectItem>
   299→                <SelectItem value="hi">हिन्दी</SelectItem>
   300→              </SelectContent>
   301→            </Select>
   302→          </div>
   303→          <div className="space-y-1.5">
   304→            <Label htmlFor="su-org" className="text-slate-300">{t('organization')} <span className="text-slate-500">({t('optional')})</span></Label>
   305→            <Input id="su-org" value={organization} onChange={(e) => setOrganization(e.target.value)} className={inputCls} />
   306→          </div>
   307→        </div>
   308→        <div className="space-y-1.5">
   309→          <Label className="text-slate-300">{t('password')}</Label>
   310→          <PasswordField value={password} onChange={setPassword} placeholder="8+ characters" show={show} setShow={setShow} autoComplete="new-password" />
   311→        </div>
   312→        <div className="space-y-1.5">
   313→          <Label className="text-slate-300">{t('confirmPassword')}</Label>

   332→// ============ Email verify (demo-labelled) ============
   333→export function VerifyScreen() {
   334→  const t = useTranslations('auth');
   335→  const { pendingEmail, pendingToken, user, setUser, setView } = useAppStore();
   336→  const [loading, setLoading] = useState(false);
   337→  const [err, setErr] = useState('');
   338→  const emailShown = pendingEmail || user?.email || '';
   339→  const canTokenVerify = !!pendingEmail && !!pendingToken;
   340→  const canSelfVerify = !!user && !user.emailVerified;
   341→
   342→  async function verify() {
   343→    setLoading(true);
   344→    setErr('');
   345→    try {
   346→      if (canTokenVerify) {
   347→        await api('/api/auth', { body: { action: 'verify', email: pendingEmail, token: pendingToken } });
   348→        const session = await api<{ user: import('@/lib/types').SessionUser }>('/api/auth');
   349→        setUser(session.user);
   350→      } else {
   351→        const res = await api<{ user: import('@/lib/types').SessionUser }>('/api/auth', { body: { action: 'verifySelf' } });
   352→        setUser(res.user);
   353→      }
   354→      setView('roleSelect');
   355→    } catch (e2) {
   356→      setErr((e2 as Error).message);
   357→    } finally {
   358→      setLoading(false);
   359→    }
   360→  }
   361→
   362→  return (
   363→    <AuthShell title={t('verifyTitle')} sub={t('verifySub')}>
   364→      <div className="space-y-4">
   365→        <ErrAlert msg={err} />
   366→        <Alert className="border-sky-500/30 bg-sky-500/10 text-sky-300">
   367→          <Info className="size-4" />
   368→          <AlertDescription className="text-[12.5px] leading-relaxed">{t('verifyDemo')}</AlertDescription>
   369→        </Alert>
   370→        <div className="rounded-lg border border-white/10 bg-white/5 p-3 text-[12px] text-slate-300">

   372→          <div className="mt-1"><span className="text-slate-500">Token:</span> <span className="font-mono break-all">{pendingToken ? pendingToken.slice(0, 8) + '…' : '—'}</span></div>

   374→        <SubmitBtn loading={loading} disabled={!canTokenVerify && !canSelfVerify} onClick={verify}>{t('verifyCta')}</SubmitBtn>
   375→      </div>
   376→    </AuthShell>

   689→export function ChangePasswordGate() {
   690→  const t = useTranslations('auth.changePassword');
   691→  const { user, setUser } = useAppStore();
   692→  const [current, setCurrent] = useState('');
   693→  const [password, setPassword] = useState('');
   694→  const [show, setShow] = useState(false);
   695→  const [loading, setLoading] = useState(false);
   696→  const [err, setErr] = useState('');
   697→
   698→  async function submit(e: React.FormEvent) {
   699→    e.preventDefault();
   700→    setLoading(true);
   701→    setErr('');
   702→    try {
   703→      await api('/api/auth', { body: { action: 'changePassword', currentPassword: current, password } });
   704→      const session = await api<{ user: import('@/lib/types').SessionUser }>('/api/auth');
   705→      setUser(session.user);
   706→    } catch (e2) {
   707→      setErr((e2 as Error).message);

   709→      setLoading(false);
   710→    }
   711→  }
   712→
   713→  return (
   714→    <AuthShell title={t('title')} sub={t('sub')}>
   715→      <form onSubmit={submit} className="space-y-4">

   722→          <Label className="text-slate-300">{t('current')}</Label>

   724→        </div>
   725→        <div className="space-y-1.5">
   726→          <Label className="text-slate-300">{t('newPassword')}</Label>

   730→      </form>
   731→    </AuthShell>
