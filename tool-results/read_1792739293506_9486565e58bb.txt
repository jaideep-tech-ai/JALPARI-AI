    53→/** deterministic pseudo-telemetry from an id (demo-labelled chips) */
    54→function hashNum(id: string, min: number, max: number): number {
    55→  let h = 0;
    56→  for (let i = 0; i < id.length; i++) h = (h * 31 + id.charCodeAt(i)) | 0;
    57→  const n = Math.abs(h) % 100;
    58→  return min + Math.round((n / 100) * (max - min));
    59→}

    61→// ============ 3D HYDRO TERRAIN VIEWPORT ============
    62→function HydroTerrain({ critical, rainMm }: { critical: number; rainMm: number }) {
    63→  const t = useTranslations('bento');
    64→  const planeRef = useRef<HTMLDivElement>(null);
    65→  const drag = useRef<{ on: boolean; x: number; y: number; rx: number; rz: number }>({ on: false, x: 0, y: 0, rx: 56, rz: -18 });
    66→  const [rot, setRot] = useState({ rx: 56, rz: -18 });
    67→  const [intensity, setIntensity] = useState(30);
    68→
    69→  // derive initial intensity from live data
    70→  useEffect(() => {
    71→    setIntensity(Math.min(96, Math.round(critical * 11 + rainMm * 0.5 + 18)));
    72→  }, [critical, rainMm]);
    73→
    74→  useEffect(() => {
    75→    const el = planeRef.current?.parentElement; // scene container
    76→    if (!el) return;
    77→    const down = (e: PointerEvent) => {
    78→      drag.current = { on: true, x: e.clientX, y: e.clientY, rx: rot.rx, rz: rot.rz };
    79→      el.setPointerCapture(e.pointerId);
    80→    };
    81→    const move = (e: PointerEvent) => {
    82→      if (!drag.current.on) return;
    83→      const dx = e.clientX - drag.current.x;
    84→      const dy = e.clientY - drag.current.y;
    85→      setRot({
    86→        rx: Math.max(26, Math.min(78, drag.current.rx - dy * 0.22)),
    87→        rz: Math.max(-48, Math.min(48, drag.current.rz + dx * 0.22)),
    88→      });
    89→    };
    90→    const up = () => { drag.current.on = false; };
    91→    el.addEventListener('pointerdown', down);
    92→    el.addEventListener('pointermove', move);
    93→    el.addEventListener('pointerup', up);
    94→    el.addEventListener('pointercancel', up);
    95→    return () => {
    96→      el.removeEventListener('pointerdown', down);
    97→      el.removeEventListener('pointermove', move);
    98→      el.removeEventListener('pointerup', up);
    99→      el.removeEventListener('pointercancel', up);
   100→    };
   101→  }, [rot.rx, rot.rz]);
   102→
   103→  const hot = intensity >= 55;
   104→  const waterZ = 4 + (intensity / 100) * 26;
   105→  const levels = ['CALM', 'RAIN', 'FLOOD'] as const;
   106→  const levelVal = { CALM: 18, RAIN: 52, FLOOD: 92 } as const;
   107→
   108→  return (
   109→    <div className="hc-bento hc-morph relative overflow-hidden p-4" data-testid="spline-dashboard-viewport">
   110→      {/* spline embed slot — set SPLINE_DASHBOARD_SCENE_URL to overlay */}
   111→      <div className="mb-3 flex items-center justify-between gap-2">
   112→        <span className="flex items-center gap-1.5 text-[10.5px] font-bold uppercase tracking-[0.2em] text-muted-foreground">
   113→          <Radar className="size-3.5 text-[var(--hc-accent)]" />
   114→          {t('terrainTitle')}
   115→        </span>
   116→        <SourceChip status={hot ? 'SIMULATED' : 'LIVE'} />
   117→      </div>
   118→
   119→      <div className="hc-terrain-scene relative h-[300px] cursor-grab touch-none select-none active:cursor-grabbing" style={{ ['--hc-accent' as string]: hot ? '#FF4D5E' : undefined }}>
   120→        <div
   121→          ref={planeRef}
   122→          className="hc-terrain-plane absolute left-1/2 top-1/2 h-[300px] w-[300px] -translate-x-1/2 -translate-y-1/2"
   123→          style={{ transform: `translate(-50%, -50%) rotateX(${rot.rx}deg) rotateZ(${rot.rz}deg)` }}
   124→        >
   125→          {/* terrain slab */}
   126→          <div
   127→            className="absolute inset-0 rounded-[22px] border border-white/10"
   128→            style={{
   129→              background: `
   130→                radial-gradient(120px 90px at 26% 30%, rgba(16,217,160,0.16), transparent 70%),
   131→                radial-gradient(150px 110px at 74% 66%, rgba(0,242,254,0.13), transparent 70%),
   132→                radial-gradient(90px 70px at 60% 22%, rgba(167,139,250,0.10), transparent 70%),
   133→                linear-gradient(145deg, rgba(14,22,38,0.9), rgba(6,10,19,0.95))`,
   134→              boxShadow: 'inset 0 0 40px rgba(0,0,0,0.55), 0 30px 60px -18px rgba(0,0,0,0.7)',
   135→            }}
   136→          >
   137→            {/* street grid */}
   138→            <div
   139→              className="absolute inset-0 rounded-[22px] opacity-35"
   140→              style={{
   141→                backgroundImage:
   142→                  'linear-gradient(rgba(255,255,255,0.16) 1px, transparent 1px), linear-gradient(90deg, rgba(255,255,255,0.16) 1px, transparent 1px)',
   143→                backgroundSize: '30px 30px',
   144→                borderRadius: 'inherit',
   145→                maskImage: 'radial-gradient(circle at 50% 50%, black 40%, transparent 95%)',
   146→              }}
   147→            />
   148→            {/* river channel */}
   149→            <svg viewBox="0 0 300 300" className="absolute inset-0 h-full w-full opacity-70">
   150→              <path d="M-8 210 C 60 196, 96 236, 150 218 S 246 178, 310 196" fill="none" stroke="rgba(0,242,254,0.5)" strokeWidth="7" strokeLinecap="round" />
   151→              <path d="M-8 210 C 60 196, 96 236, 150 218 S 246 178, 310 196" fill="none" stroke="rgba(0,242,254,0.22)" strokeWidth="16" strokeLinecap="round" />
   152→            </svg>
   153→
   154→            {/* liquid water layer — rises & shifts colour with intensity */}
   155→            <div
   156→              className={cn('hc-water-layer absolute inset-1 rounded-[18px]', hot && 'hc-water-hot')}
   157→              style={{ transform: `translateZ(${waterZ}px)`, opacity: 0.34 + (intensity / 100) * 0.4 }}
   158→            />
   159→
   160→            {/* vertical beacons (hotspots) */}
   161→            {[
   162→              { x: '24%', y: '30%', h: 34 + intensity * 0.3, c: hot ? '#FF4D5E' : '#00F2FE' },
   163→              { x: '68%', y: '58%', h: 22 + intensity * 0.22, c: hot ? '#FFB224' : '#10D9A0' },
   164→              { x: '46%', y: '74%', h: 16 + intensity * 0.16, c: '#A78BFA' },
   165→            ].map((b, i) => (
   166→              <div key={i} className="absolute" style={{ left: b.x, top: b.y, transform: `translateZ(${waterZ + 2}px)`, transformStyle: 'preserve-3d' }}>
   167→                <div className="hc-beacon relative w-[3px]" style={{ height: b.h, transform: 'rotateX(-90deg)', transformOrigin: 'bottom center' }}>
   168→                  <span className="absolute -top-[5px] left-1/2 size-[9px] -translate-x-1/2 rounded-full hc-blink-dot" style={{ background: b.c, boxShadow: `0 0 14px ${b.c}` }} />
   169→                </div>
   170→              </div>
   171→            ))}
   172→          </div>
   173→
   174→          {/* ground glow */}
   175→          <div className="absolute left-1/2 top-1/2 h-[280px] w-[280px] -translate-x-1/2 -translate-y-1/2 rounded-[100%] blur-2xl"
   176→            style={{ background: hot ? 'rgba(255,77,94,0.16)' : 'rgba(0,242,254,0.14)', transform: `translate(-50%,-50%) rotateX(90deg) translateZ(-90px)` }} />
   177→        </div>
   178→
   179→        {/* HUD */}
   180→        <div className="pointer-events-none absolute left-3 top-3 rounded-lg border border-white/10 bg-black/35 px-2 py-1 backdrop-blur-md">
   181→          <span className="flex items-center gap-1.5 text-[10px] font-medium text-white/70">
   182→            <Move3d className="size-3" /> {t('dragHint')}
   183→          </span>
   184→        </div>
   185→        <div className="absolute bottom-3 right-3 rounded-lg border border-white/10 bg-black/35 px-2.5 py-1 backdrop-blur-md">
   186→          <span className={cn('font-mono text-[13px] font-bold', hot ? 'text-[#FF4D5E]' : 'hc-num')}>{intensity}%</span>
   187→        </div>
   188→      </div>
   189→
   190→      {/* water-level control */}
   191→      <div className="mt-3 grid grid-cols-3 gap-1.5">
   192→        {levels.map((lv) => (
   193→          <button
   194→            key={lv}
   195→            type="button"
   196→            onClick={() => setIntensity(levelVal[lv])}
   197→            className={cn(
   198→              'hc-focusable rounded-lg border px-2 py-1.5 text-[10.5px] font-bold uppercase tracking-[0.14em] transition-all duration-300',
   199→              Math.abs(intensity - levelVal[lv]) < 12
   200→                ? 'border-[var(--hc-accent)]/60 bg-[var(--hc-accent)]/10 text-[var(--hc-accent)]'
   201→                : 'border-white/10 text-muted-foreground hover:border-white/25 hover:text-foreground'
   202→            )}
   203→          >
   204→            {t(`level.${lv.toLowerCase()}`)}
   205→          </button>
   206→        ))}
   207→      </div>
   208→    </div>
   209→  );
   210→}
