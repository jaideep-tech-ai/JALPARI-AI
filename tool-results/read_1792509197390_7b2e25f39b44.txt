    79→// ============ Password field with strength + visibility ============
    80→function PasswordField({
    81→  value, onChange, placeholder, show, setShow, autoComplete,
    82→}: {
    83→  value: string; onChange: (v: string) => void; placeholder: string;
    84→  show: boolean; setShow: (v: boolean) => void; autoComplete?: string;
    85→}) {
    86→  const t = useTranslations('auth');
    87→  const strength = value.length >= 12 && /[A-Z]/.test(value) && /[0-9]/.test(value) ? 4
    88→    : value.length >= 8 && /[A-Za-z]/.test(value) && /[0-9]/.test(value) ? 3
    89→    : value.length >= 8 ? 2 : value.length >= 4 ? 1 : 0;
    90→  const labels = [t('pwdWeak'), t('pwdWeak'), t('pwdFair'), t('pwdGood'), t('pwdStrong')];
    91→  const colors = ['bg-red-500', 'bg-red-500', 'bg-amber-500', 'bg-sky-500', 'bg-[#10D9A0]'];
    92→  return (
    93→    <div>
    94→      <div className="hc-field">
    95→        <input
    96→          type={show ? 'text' : 'password'}
    97→          value={value}
    98→          onChange={(e) => onChange(e.target.value)}
    99→          placeholder=" "
   100→          autoComplete={autoComplete}
   101→          className="hc-input pr-11"
   102→        />
   103→        <label className="hc-label">{placeholder}</label>
   104→        <button
   105→          type="button"
   106→          onClick={() => setShow(!show)}
   107→          className="absolute right-3 top-[15px] text-slate-500 transition-colors hover:text-white"
   108→          aria-label={show ? t('hidePassword') : t('showPassword')}
   109→        >
   110→          {show ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   111→        </button>
   112→      </div>
   113→      {value && (
   114→        <div className="mt-1.5 flex items-center gap-2">
   115→          <div className="flex gap-1">
   116→            {[1, 2, 3, 4].map((i) => (
   117→              <span key={i} className={`h-1 w-8 rounded-full ${i <= strength ? colors[strength] : 'bg-white/10'}`} />
   118→            ))}
   119→          </div>
   120→          <span className="text-[11px] text-slate-500">{labels[strength]}</span>
   121→        </div>
   122→      )}
   123→    </div>
   124→  );
   125→}

   351→// ============ Sign up ============
   352→export function SignUpScreen() {
   353→  const t = useTranslations('auth');
   354→  const tv = useTranslations('auth.validation');
   355→  const { setView, setPending } = useAppStore();
   356→  const [name, setName] = useState('');
   357→  const [email, setEmail] = useState('');
   358→  const [password, setPassword] = useState('');
   359→  const [confirm, setConfirm] = useState('');
   360→  const [language, setLanguage] = useState<'en' | 'hi'>('en');
   361→  const [organization, setOrganization] = useState('');
   362→  const [terms, setTerms] = useState(false);
   363→  const [show, setShow] = useState(false);
   364→  const [loading, setLoading] = useState(false);
   365→  const [err, setErr] = useState('');
   366→
   367→  async function submit(e: React.FormEvent) {
   368→    e.preventDefault();
   369→    setErr('');
   370→    if (name.trim().length < 2) return setErr(tv('nameMin'));
   371→    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return setErr(tv('emailInvalid'));
   372→    if (password.length < 8) return setErr(tv('pwdMin'));
   373→    if (password !== confirm) return setErr(tv('pwdMatch'));
   374→    if (!terms) return setErr(tv('termsRequired'));
   375→    setLoading(true);
   376→    try {
   377→      const res = await api<{ demoVerifyToken: string }>('/api/auth', {
   378→        body: { action: 'signup', name, email, password, language, organization },
   379→      });
   380→      setPending({ email, token: res.demoVerifyToken });
   381→      setView('verify');
   382→    } catch (e2) {
   383→      setErr((e2 as Error).message);
   384→    } finally {
   385→      setLoading(false);
   386→    }
   387→  }
   388→
   389→  return (
   390→    <AuthSplit active="signup" title={t('signUpTitle')} sub={t('signUpSub')}>
   391→      <form onSubmit={submit} className="space-y-4">
   392→        <ErrAlert msg={err} />
   393→        <HydroInput id="su-name" label={t('fullName')} value={name} onChange={setName} autoComplete="name" required />
   394→        <HydroInput id="su-email" label={t('email')} type="email" value={email} onChange={setEmail} autoComplete="email" required />
   395→        <div className="grid gap-4 sm:grid-cols-2">
   396→          <div className="space-y-1.5">
   397→            <Label className="text-[12.5px] text-slate-400">{t('preferredLanguage')}</Label>
   398→            <Select value={language} onValueChange={(v) => setLanguage(v as 'en' | 'hi')}>
   399→              <SelectTrigger className={inputCls}><SelectValue /></SelectTrigger>
   400→              <SelectContent>
   401→                <SelectItem value="en">English</SelectItem>
   402→                <SelectItem value="hi">हिन्दी</SelectItem>
   403→              </SelectContent>
   404→            </Select>
   405→          </div>
   406→          <div className="hc-field">
   407→            <input
   408→              id="su-org"
   409→              value={organization}
   410→              onChange={(e) => setOrganization(e.target.value)}
   411→              placeholder=" "
   412→              autoComplete="organization"
   413→              className="hc-input"
   414→            />
   415→            <label htmlFor="su-org" className="hc-label">
   416→              {t('organization')} · {t('optional')}
   417→            </label>
   418→          </div>
   419→        </div>
   420→        <PasswordField value={password} onChange={setPassword} placeholder={t('password')} show={show} setShow={setShow} autoComplete="new-password" />
   421→        <HydroInput id="su-confirm" label={t('confirmPassword')} type="password" value={confirm} onChange={setConfirm} autoComplete="new-password" required />
   422→        <label className="flex items-start gap-2.5 text-[13px] text-slate-400">
   423→          <Checkbox checked={terms} onCheckedChange={(v) => setTerms(v === true)} className="mt-0.5 border-white/30" />
   424→          {t('terms')}
   425→        </label>
   426→        <SubmitBtn loading={loading}>{t('signUpCta')}</SubmitBtn>
   427→      </form>
   428→    </AuthSplit>
   429→  );
   430→}
