    79→// ============ Password field with strength + visibility ============
    80→function PasswordField({
    81→  value, onChange, placeholder, show, setShow, autoComplete,
    82→}: {
    83→  value: string; onChange: (v: string) => void; placeholder: string;
    84→  show: boolean; setShow: (v: boolean) => void; autoComplete?: string;
    85→}) {
    86→  const t = useTranslations('auth');
    87→  const strength = value.length >= 12 && /[A-Z]/.test(value) && /[0-9]/.test(value) ? 4
    88→    : value.length >= 8 && /[A-Za-z]/.test(value) && /[0-9]/.test(value) ? 3
    89→    : value.length >= 8 ? 2 : value.length >= 4 ? 1 : 0;
    90→  const labels = [t('pwdWeak'), t('pwdWeak'), t('pwdFair'), t('pwdGood'), t('pwdStrong')];
    91→  const colors = ['bg-red-500', 'bg-red-500', 'bg-amber-500', 'bg-sky-500', 'bg-[#10D9A0]'];
    92→  return (
    93→    <div>
    94→      <div className="hc-field">
    95→        <input
    96→          type={show ? 'text' : 'password'}
    97→          value={value}
    98→          onChange={(e) => onChange(e.target.value)}
    99→          placeholder=" "
   100→          autoComplete={autoComplete}
   101→          className="hc-input pr-11"
   102→        />
   103→        <label className="hc-label">{placeholder}</label>
   104→        <button
   105→          type="button"
   106→          onClick={() => setShow(!show)}
   107→          className="absolute right-3 top-[15px] text-slate-500 transition-colors hover:text-white"
   108→          aria-label={show ? t('hidePassword') : t('showPassword')}
   109→        >
   110→          {show ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   111→        </button>
   112→      </div>
   113→      {value && (
   114→        <div className="mt-1.5 flex items-center gap-2">
   115→          <div className="flex gap-1">
   116→            {[1, 2, 3, 4].map((i) => (
   117→              <span key={i} className={`h-1 w-8 rounded-full ${i <= strength ? colors[strength] : 'bg-white/10'}`} />
   118→            ))}
   119→          </div>
   120→          <span className="text-[11px] text-slate-500">{labels[strength]}</span>
   121→        </div>
   122→      )}
   123→    </div>
   124→  );
   125→}

   353→// ============ Sign up ============
   354→export function SignUpScreen() {
   355→  const t = useTranslations('auth');
   356→  const tv = useTranslations('auth.validation');
   357→  const { setView, setPending } = useAppStore();
   358→  const [name, setName] = useState('');
   359→  const [email, setEmail] = useState('');
   360→  const [password, setPassword] = useState('');
   361→  const [confirm, setConfirm] = useState('');
   362→  const [language, setLanguage] = useState<'en' | 'hi'>('en');
   363→  const [organization, setOrganization] = useState('');
   364→  const [terms, setTerms] = useState(false);
   365→  const [show, setShow] = useState(false);
   366→  const [loading, setLoading] = useState(false);
   367→  const [err, setErr] = useState('');
   368→
   369→  async function submit(e: React.FormEvent) {
   370→    e.preventDefault();
   371→    setErr('');
   372→    if (name.trim().length < 2) return setErr(tv('nameMin'));
   373→    if (!/^[^@\s]+@[^@\s]+\.[^@\s]+$/.test(email)) return setErr(tv('emailInvalid'));
   374→    if (password.length < 8) return setErr(tv('pwdMin'));
   375→    if (password !== confirm) return setErr(tv('pwdMatch'));
   376→    if (!terms) return setErr(tv('termsRequired'));
   377→    setLoading(true);
   378→    try {
   379→      const res = await api<{ demoVerifyToken: string }>('/api/auth', {
   380→        body: { action: 'signup', name, email, password, language, organization },
   381→      });
   382→      setPending({ email, token: res.demoVerifyToken });
   383→      setView('verify');
   384→    } catch (e2) {
   385→      setErr((e2 as Error).message);
   386→    } finally {
   387→      setLoading(false);
   388→    }
   389→  }
   390→
   391→  return (
   392→    <AuthSplit active="signup" title={t('signUpTitle')} sub={t('signUpSub')}>
   393→      <form onSubmit={submit} className="space-y-4">
   394→        <ErrAlert msg={err} />
   395→        <HydroInput id="su-name" label={t('fullName')} value={name} onChange={setName} autoComplete="name" required />
   396→        <HydroInput id="su-email" label={t('email')} type="email" value={email} onChange={setEmail} autoComplete="email" required />
   397→        <div className="grid gap-4 sm:grid-cols-2">
   398→          <div className="space-y-1.5">
   399→            <Label className="text-[12.5px] text-slate-400">{t('preferredLanguage')}</Label>
   400→            <Select value={language} onValueChange={(v) => setLanguage(v as 'en' | 'hi')}>
   401→              <SelectTrigger className={inputCls}><SelectValue /></SelectTrigger>
   402→              <SelectContent>
   403→                <SelectItem value="en">English</SelectItem>
   404→                <SelectItem value="hi">हिन्दी</SelectItem>
   405→              </SelectContent>
   406→            </Select>
   407→          </div>
   408→          <div className="hc-field">
   409→            <input
   410→              id="su-org"
   411→              value={organization}
   412→              onChange={(e) => setOrganization(e.target.value)}
   413→              placeholder=" "
   414→              autoComplete="organization"
   415→              className="hc-input"
   416→            />
   417→            <label htmlFor="su-org" className="hc-label">
   418→              {t('organization')} · {t('optional')}
   419→            </label>
   420→          </div>
   421→        </div>
   422→        <PasswordField value={password} onChange={setPassword} placeholder={t('password')} show={show} setShow={setShow} autoComplete="new-password" />
   423→        <HydroInput id="su-confirm" label={t('confirmPassword')} type="password" value={confirm} onChange={setConfirm} autoComplete="new-password" required />
   424→        <label className="flex items-start gap-2.5 text-[13px] text-slate-400">
   425→          <Checkbox checked={terms} onCheckedChange={(v) => setTerms(v === true)} className="mt-0.5 border-white/30" />
   426→          {t('terms')}
   427→        </label>
   428→        <SubmitBtn loading={loading}>{t('signUpCta')}</SubmitBtn>
   429→      </form>
   430→    </AuthSplit>
   431→  );
   432→}
