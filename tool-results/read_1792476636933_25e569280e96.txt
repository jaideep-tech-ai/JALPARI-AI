    80→    80→function PasswordField({
    81→    81→  value, onChange, placeholder, show, setShow, autoComplete,
    82→    82→}: {
    83→    83→  value: string; onChange: (v: string) => void; placeholder: string;
    84→    84→  show: boolean; setShow: (v: boolean) => void; autoComplete?: string;
    85→    85→}) {
    86→    86→  const t = useTranslations('auth');
    87→    87→  const strength = value.length >= 12 && /[A-Z]/.test(value) && /[0-9]/.test(value) ? 4
    88→    88→    : value.length >= 8 && /[A-Za-z]/.test(value) && /[0-9]/.test(value) ? 3
    89→    89→    : value.length >= 8 ? 2 : value.length >= 4 ? 1 : 0;
    90→    90→  const labels = [t('pwdWeak'), t('pwdWeak'), t('pwdFair'), t('pwdGood'), t('pwdStrong')];
    91→    91→  const colors = ['bg-red-500', 'bg-red-500', 'bg-amber-500', 'bg-sky-500', 'bg-emerald-500'];
    92→    92→  return (
    93→    93→    <div>
    94→    94→      <div className="relative">
    95→    95→        <Input
    96→    96→          type={show ? 'text' : 'password'}
    97→    97→          value={value}
    98→    98→          onChange={(e) => onChange(e.target.value)}
    99→    99→          placeholder={placeholder}
   100→   100→          autoComplete={autoComplete}
   101→   101→          className={inputCls + ' pr-10'}
   102→   102→        />
   103→   103→        <button
   104→   104→          type="button"
   105→   105→          onClick={() => setShow(!show)}
   106→   106→          className="absolute right-2.5 top-2.5 text-slate-400 hover:text-white"
   107→   107→          aria-label={show ? t('hidePassword') : t('showPassword')}
   108→   108→        >
   109→   109→          {show ? <EyeOff className="size-4" /> : <Eye className="size-4" />}
   110→   110→        </button>
   111→   111→      </div>
   112→   112→      {value && (
   113→   113→        <div className="mt-1.5 flex items-center gap-2">
   114→   114→          <div className="flex gap-1">
   115→   115→            {[1, 2, 3, 4].map((i) => (
   116→   116→              <span key={i} className={`h-1 w-8 rounded-full ${i <= strength ? colors[strength] : 'bg-white/10'}`} />
   117→   117→            ))}
   118→   118→          </div>
   119→   119→          <span className="text-[11px] text-slate-400">{labels[strength]}</span>
   120→   120→        </div>
   121→   121→      )}
   122→   122→    </div>
   123→   123→  );
   124→   124→}
