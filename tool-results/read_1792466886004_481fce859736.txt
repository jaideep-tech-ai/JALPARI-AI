    23→function Reveal({ children, delay = 0, className }: { children: React.ReactNode; delay?: number; className?: string }) {
    24→  const reduce = useReducedMotion();
    25→  return (
    26→    <motion.div
    27→      className={className}
    28→      initial={reduce ? false : { opacity: 0, y: 26 }}
    29→      whileInView={{ opacity: 1, y: 0 }}
    30→      viewport={{ once: true, margin: '-60px' }}
    31→      transition={{ duration: 0.65, delay, ease: [0.22, 1, 0.36, 1] }}
    32→    >
    33→      {children}
    34→    </motion.div>
    35→  );
    36→}

    38→function Kicker({ children, idx }: { children: React.ReactNode; idx?: string }) {

    56→}

    58→/** card with mouse-tracked radial spotlight */
    59→function SpotlightCard({ children, className }: { children: React.ReactNode; className?: string }) {
    60→  return (
    61→    <div
    62→      className={`jg-spot jg-card-hover h-full rounded-xl border bg-card ${className ?? ''}`}
    63→      onMouseMove={(e) => {
    64→        const r = e.currentTarget.getBoundingClientRect();
    65→        e.currentTarget.style.setProperty('--mx', `${e.clientX - r.left}px`);
    66→        e.currentTarget.style.setProperty('--my', `${e.clientY - r.top}px`);
    67→      }}
    68→    >
    69→      {children}
    70→    </div>
    71→  );
    72→}
