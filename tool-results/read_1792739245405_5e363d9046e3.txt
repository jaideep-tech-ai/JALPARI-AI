   460→function FieldDashboard() {
   461→  const t = useTranslations('app.field');
   462→  const tt = useTranslations('app.tasks');
   463→  const [tasks, setTasks] = useState<Task[]>([]);
   464→  const [loading, setLoading] = useState(true);
   465→  const [selected, setSelected] = useState<Task | null>(null);
   466→
   467→  async function load() {
   468→    setLoading(true);
   469→    try {
   470→      const r = await api<{ tasks: Task[] }>('/api/tasks?mine=1');
   471→      setTasks(r.tasks);
   472→    } finally {
   473→      setLoading(false);
   474→    }
   475→  }
   476→  useEffect(() => { load(); }, []);
   477→
   478→  const active = tasks.filter((tk) => ['ASSIGNED', 'IN_PROGRESS'].includes(tk.status));
   479→  const submitted = tasks.filter((tk) => tk.status === 'SUBMITTED');
   480→  const done = tasks.filter((tk) => ['APPROVED', 'DONE'].includes(tk.status));
   481→
   482→  return (
   483→    <div className="space-y-4">
   484→      <h1 className="text-lg font-bold">{t('title')}</h1>
   485→      <div className="grid grid-cols-3 gap-3">
   486→        <StatCard icon={HardHat} label="Active" value={active.length} tone="border-sky-500/25 bg-sky-500/10 text-sky-500" />
   487→        <StatCard icon={FileCheck2} label="Awaiting verification" value={submitted.length} tone="border-violet-500/25 bg-violet-500/10 text-violet-500" />
   488→        <StatCard icon={ShieldCheck} label="Verified" value={done.length} tone="border-emerald-500/25 bg-emerald-500/10 text-emerald-500" />
   489→      </div>
   490→
   491→      {loading ? (
   492→        <div className="grid gap-3 md:grid-cols-2">{[1, 2, 3, 4].map((i) => <Skeleton key={i} className="h-28 rounded-xl" />)}</div>
   493→      ) : tasks.length === 0 ? (
   494→        <Card>
   495→          <CardContent className="flex flex-col items-center gap-2 py-12 text-center">
   496→            <HardHat className="size-8 text-muted-foreground/50" />
   497→            <p className="text-sm text-muted-foreground">{t('noTasks')}</p>
   498→          </CardContent>
   499→        </Card>
   500→      ) : (
   501→        <div className="grid gap-3 md:grid-cols-2">
   502→          {[...active, ...submitted, ...done].map((tk) => (
   503→            <button key={tk.id} onClick={() => setSelected(tk)} className="text-left">
   504→              <Card className="jg-card-hover h-full py-4">
   505→                <CardContent className="space-y-2 px-4">
   506→                  <div className="text-[13.5px] font-semibold leading-snug">{tk.title}</div>
   507→                  <div className="flex flex-wrap items-center gap-1.5">
   508→                    <StatusPill status={tk.status} />
   509→                    <PriorityBadge priority={tk.priority as 'MEDIUM'} />
   510→                    {tk.dueAt && <Badge variant="outline" className="text-[9px]">{t('dueLabel')}: {new Date(tk.dueAt).toLocaleDateString()}</Badge>}
   511→                  </div>
   512→                  <div className="text-[10.5px] text-muted-foreground">
   513→                    {tt('evidenceCount', { n: JSON.parse(tk.evidence || '[]').length })}
   514→                    {tk.isDemo && ' · DEMO'}
   515→                  </div>
   516→                </CardContent>
   517→              </Card>
   518→            </button>
   519→          ))}
   520→        </div>
   521→      )}
   522→      <TaskDetailDialog task={selected} onClose={() => setSelected(null)} onChanged={load} />
   523→    </div>
   524→  );
   525→}

   528→function AdminDashboard() {
   529→  const t = useTranslations('app.admin');
   530→  const ta = useTranslations('app.nav');
   531→  const setAppPage = useAppStore((s) => s.setAppPage);
   532→  const [pending, setPending] = useState<{ id: string; requestedRole: string; user: { name: string; email: string }; createdAt: string; organization: string | null }[]>([]);
   533→  const [users, setUsers] = useState<{ id: string; role: string | null; email: string; isDemo: boolean }[]>([]);
   534→  const [sources, setSources] = useState<{ key: string; name: string; provider: string; status: string; notes: string | null }[]>([]);
   535→  const [busyId, setBusyId] = useState<string | null>(null);
   536→
   537→  async function load() {
   538→    const u = await api<{ users: typeof users; pending: typeof pending }>('/api/admin?section=users').catch(() => ({ users: [], pending: [] }));
   539→    setUsers(u.users);
   540→    setPending(u.pending);
   541→    const s = await api<{ sources: typeof sources }>('/api/admin?section=datasources').catch(() => ({ sources: [] }));
   542→    setSources(s.sources);
   543→  }
   544→  useEffect(() => { load(); }, []);
   545→
   546→  async function decide(id: string, decision: 'approve' | 'reject') {
   547→    setBusyId(id);
   548→    try {
   549→      await api('/api/admin', { body: { action: 'decideRole', id, decision } });
   550→      toast({ title: decision === 'approve' ? t('roleApproved') : t('roleRejected') });
   551→      await load();
   552→    } catch (e) {
   553→      toast({ title: (e as Error).message, variant: 'destructive' });
   554→    } finally {
   555→      setBusyId(null);
   556→    }
   557→  }
   558→
   559→  return (
   560→    <div className="space-y-4">
   561→      <h1 className="text-lg font-bold">{t('title')}</h1>
   562→      <div className="grid grid-cols-2 gap-3 lg:grid-cols-4">
   563→        <StatCard icon={Users} label={t('users')} value={users.length} tone="border-sky-500/25 bg-sky-500/10 text-sky-500" />
   564→        <StatCard icon={ShieldQuestion} label={t('roleRequests')} value={pending.length} tone="border-amber-500/25 bg-amber-500/10 text-amber-500" />
   565→        <StatCard icon={Database} label={t('sourcesHealth')} value={sources.filter((s) => s.status !== 'UNAVAILABLE').length + '/' + sources.length} tone="border-emerald-500/25 bg-emerald-500/10 text-emerald-500" />
   566→        <StatCard icon={Radio} label={t('appHealth')} value="healthy" tone="border-emerald-500/25 bg-emerald-500/10 text-emerald-500" />
   567→      </div>
   568→
   569→      <Card className="py-4">
   570→        <CardHeader className="flex-row items-center justify-between px-4 pb-0">
   571→          <CardTitle className="text-[13px] font-semibold">{t('roleRequests')}</CardTitle>
   572→          <Button size="sm" variant="ghost" className="h-7 text-[11px]" onClick={() => setAppPage('adminUsers')}>
   573→            {ta('adminUsers')} <ChevronRight className="size-3" />
   574→          </Button>
   575→        </CardHeader>
   576→        <CardContent className="px-4 pt-2">
   577→          {pending.length === 0 ? (
   578→            <div className="py-6 text-center text-xs text-muted-foreground">{t('noPending')}</div>
   579→          ) : (
   580→            <div className="space-y-2">
   581→              {pending.map((p) => (
   582→                <div key={p.id} className="flex flex-wrap items-center gap-2 rounded-lg border bg-muted/30 p-3">
   583→                  <div className="min-w-0 flex-1">
   584→                    <div className="text-[12.5px] font-semibold">{p.user?.name ?? p.id}</div>
   585→                    <div className="text-[11px] text-muted-foreground">{p.user?.email} → <span className="font-mono uppercase">{p.requestedRole}</span> · {timeAgo(p.createdAt)}</div>
   586→                  </div>
   587→                  <Button size="sm" className="h-7 bg-emerald-500 text-[11px] font-semibold text-[#06251B] hover:bg-emerald-400" disabled={busyId === p.id} onClick={() => decide(p.id, 'approve')}>
   588→                    {t('approve')}
   589→                  </Button>
   590→                  <Button size="sm" variant="outline" className="h-7 border-red-500/40 text-[11px] text-red-500" disabled={busyId === p.id} onClick={() => decide(p.id, 'reject')}>
   591→                    {t('reject')}
   592→                  </Button>
   593→                </div>
   594→              ))}
   595→            </div>
   596→          )}
   597→        </CardContent>
   598→      </Card>
   599→
   600→      <Card className="py-4">
   601→        <CardHeader className="px-4 pb-0">
   602→          <CardTitle className="text-[13px] font-semibold">{t('sourcesHealth')}</CardTitle>
   603→        </CardHeader>
   604→        <CardContent className="px-4 pt-2">
   605→          <div className="divide-y">
   606→            {sources.map((s) => (
   607→              <div key={s.key} className="flex items-center gap-3 py-2.5">
   608→                <div className="min-w-0 flex-1">
   609→                  <div className="text-[12.5px] font-medium">{s.name} <span className="text-[10px] text-muted-foreground">· {s.provider}</span></div>
   610→                  {s.notes && <div className="truncate text-[10.5px] text-muted-foreground">{s.notes}</div>}
   611→                </div>
   612→                <SourceChip status={s.status as 'LIVE'} />
   613→              </div>
   614→            ))}
   615→          </div>

   619→      <Card className="py-4">
   620→        <CardHeader className="px-4 pb-0">
   621→          <CardTitle className="text-[13px] font-semibold">{t('bootstrap')}</CardTitle>
   622→        </CardHeader>
